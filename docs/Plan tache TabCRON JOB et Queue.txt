Job Queue & Cron Scheduler - Implementation Plan
Executive Summary
This plan outlines the implementation of a comprehensive asynchronous job processing and cron scheduling system for the LathDevinci Framework. The system will support:

Queue System: Async job dispatch with multiple drivers (Database, Redis, File)
Cron Scheduler: Periodic task execution with cron expression support
Multi-driver Architecture: Pluggable backends for different deployment scenarios
Module Integration: Standard contracts for external modules to register jobs/cron tasks
CLI Workers: Background processes for job/cron execution
Backoffice UI: Management interface for monitoring and control
Architecture Inspiration: Laravel Queue, NestJS Schedule, Symfony Messenger, BullMQ

1. Database Schema
		Table: jobs
		Stores pending/processing jobs.

		CREATE TABLE jobs (
			id BIGINT AUTO_INCREMENT PRIMARY KEY,
			queue VARCHAR(255) NOT NULL DEFAULT 'default',
			payload LONGTEXT NOT NULL,
			attempts TINYINT UNSIGNED NOT NULL DEFAULT 0,
			reserved_at INT UNSIGNED NULL,
			available_at INT UNSIGNED NOT NULL,
			created_at INT UNSIGNED NOT NULL,
			INDEX(queue),
			INDEX(reserved_at)
		);
		Table: failed_jobs
		Stores failed jobs with error details.

		CREATE TABLE failed_jobs (
			id BIGINT AUTO_INCREMENT PRIMARY KEY,
			connection VARCHAR(255) NOT NULL,
			queue VARCHAR(255) NOT NULL,
			payload LONGTEXT NOT NULL,
			exception LONGTEXT NOT NULL,
			failed_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
			INDEX(queue)
		);
		Table: cron_tasks
		Registered cron tasks from modules.

		CREATE TABLE cron_tasks (
			id INT AUTO_INCREMENT PRIMARY KEY,
			name VARCHAR(255) NOT NULL UNIQUE,
			module VARCHAR(255) NOT NULL,
			class VARCHAR(255) NOT NULL,
			expression VARCHAR(255) NOT NULL,
			description TEXT NULL,
			enabled TINYINT(1) NOT NULL DEFAULT 1,
			last_run_at DATETIME NULL,
			next_run_at DATETIME NULL,
			created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
			updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
			INDEX(enabled),
			INDEX(next_run_at)
		);
		Table: cron_logs
		Execution history for cron tasks.

		CREATE TABLE cron_logs (
			id BIGINT AUTO_INCREMENT PRIMARY KEY,
			task_id INT NOT NULL,
			started_at DATETIME NOT NULL,
			finished_at DATETIME NULL,
			status ENUM('running', 'success', 'failed') NOT NULL DEFAULT 'running',
			output TEXT NULL,
			error TEXT NULL,
			duration_ms INT NULL,
			created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
			FOREIGN KEY (task_id) REFERENCES cron_tasks(id) ON DELETE CASCADE,
			INDEX(task_id),
			INDEX(status)
		);

			class QueueManager
			{
				private static ?QueueManager $instance = null;
				private array $config;
				private ?QueueDriverContract $driver = null;
				public static function getInstance(): QueueManager
				{
					if (self::$instance === null) {
						self::$instance = new self();
					}
					return self::$instance;
				}
				private function __construct()
				{
					$this->loadConfig();
					$this->initializeDriver();
				}
				private function loadConfig(): void
				{
					// Load from config/queue.php
					$configPath = __DIR__ . '/../../config/queue.php';
					$this->config = file_exists($configPath) ? require $configPath : $this->getDefaultConfig();
				}
				private function initializeDriver(): void
				{
					$driver = $this->config['driver'] ?? 'database';
					
					$this->driver = match ($driver) {
						'redis' => new Drivers\RedisDriver($this->config),
						'file' => new Drivers\FileDriver($this->config),
						default => new Drivers\DatabaseDriver($this->config),
					};
				}
				public function push(string $job, array $data = [], string $queue = 'default'): bool
				{
					$payload = JobSerializer::serialize($job, $data);
					return $this->driver->push($queue, $payload);
				}
				public function getDriver(): QueueDriverContract
				{
					return $this->driver;
				}
			}

		namespace App\Core\Cron;

		class CronScheduler
		{
			private static ?CronScheduler $instance = null;

			public static function getInstance(): CronScheduler
			{
				if (self::$instance === null) {
					self::$instance = new self();

			}
			private function __construct()
			{
				$this->registry = new TaskRegistry();
			}
			/**
			 * Register a cron task.
			 */
			public function register(string $name, string $class, string $expression, string $module, string $description = ''): void
			{
				$this->registry->register($name, $class, $expression, $module, $description);
			}
			/**
			 * Run all due tasks.
			 */
			public function runDueTasks(): void
			{
				$tasks = $this->registry->getDueTasks();
				
				foreach ($tasks as $task) {
					$runner = new CronRunner($task);
					$runner->execute();
				}
			}
		}
		CronExpression.php (Parser)
		<?php
		namespace App\Core\Cron;
		class CronExpression
		{
			private string $expression;
			public function __construct(string $expression)
			{
				$this->expression = $expression;
			}
			/**
			 * Determine if the expression is due to run at the given time.
			 */
			public function isDue(\DateTime $currentTime): bool
			{
				$parts = explode(' ', $this->expression);
				
				if (count($parts) !== 5) {
					throw new \InvalidArgumentException("Invalid cron expression: {$this->expression}");
				}
				[$minute, $hour, $day, $month, $dayOfWeek] = $parts;
				return $this->matchesPart($minute, $currentTime->format('i'))
					&& $this->matchesPart($hour, $currentTime->format('H'))
					&& $this->matchesPart($day, $currentTime->format('d'))
					&& $this->matchesPart($month, $currentTime->format('m'))
					&& $this->matchesPart($dayOfWeek, $currentTime->format('w'));
			}
			private function matchesPart(string $part, string $value): bool
			{
				// * matches everything
				if ($part === '*') {
					return true;
				}
				// Exact match
				if ($part === $value) {
					return true;
				}
				// Range (e.g., 1-5)
				if (str_contains($part, '-')) {
					[$min, $max] = explode('-', $part);
					return $value >= $min && $value <= $max;
				}
				// Step values (e.g., */5)
				if (str_contains($part, '/')) {
					[$range, $step] = explode('/', $part);
					if ($range === '*') {
						return (int)$value % (int)$step === 0;
					}
				}
				// List (e.g., 1,3,5)
				if (str_contains($part, ',')) {
					return in_array($value, explode(',', $part));
				}
				return false;
			}
			/**
			 * Get the next runtime.
			 */
			public function getNextRunDate(\DateTime $currentTime = null): \DateTime
			{
				$current = $currentTime ?? new \DateTime();
				$next = clone $current;
				$next->modify('+1 minute');
				// Find the next matching time (max 1 year ahead)
				for ($i = 0; $i < 525600; $i++) {
					if ($this->isDue($next)) {
						return $next;
					}
					$next->modify('+1 minute');
				}
				throw new \RuntimeException("Could not find next run date for expression: {$this->expression}");
			}
		}

		CleanupLogsTask.php
		<?php
		namespace Modules\Admin\Cron;
		use App\Core\Cron\CronTask;
		class CleanupLogsTask extends CronTask
		{
			public function expression(): string
			{
				return '0 0 * * *'; // Every day at midnight
			}
			public function description(): string
			{
				return 'Delete log files older than 30 days';
			}
			public function handle(): void
			{
				$logPath = __DIR__ . '/../../../storage/logs';
				$files = glob($logPath . '/*.log');
				$deleted = 0;
				foreach ($files as $file) {
					if (filemtime($file) < strtotime('-30 days')) {
						unlink($file);
						$deleted++;
					}
				}
				echo "✅ Deleted $deleted old log files\n";
			}
		}

16. Risks & Mitigations
		Risk	Mitigation
		Memory leaks in long-running workers	Implement memory limit checks, restart workers periodically
		Cron expression parser bugs	Extensive unit tests, validate against known cron libraries
		Race conditions (overlapping cron)	Mutex/lock mechanism using file locks or Redis
		Failed job accumulation	Auto-cleanup policy, retention limits
		Redis unavailability	Graceful fallback to database driver
